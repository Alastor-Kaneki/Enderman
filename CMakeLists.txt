cmake_minimum_required(VERSION 3.22)
project(ModelBridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(preloader_android GIT_REPOSITORY https://github.com/LiteLDev/preloader-android.git GIT_TAG 0.2.2)
FetchContent_MakeAvailable(preloader_android)
add_library(modelbridge SHARED src/ModelBridge.cpp)
target_link_libraries(modelbridge PRIVATE preloader)
set_target_properties(modelbridge PROPERTIES OUTPUT_NAME modelbridge)
